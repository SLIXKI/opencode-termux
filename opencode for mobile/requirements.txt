# ============================================================
#  opencode-mobile — prerequisites (Termux / Android)
# ============================================================
#  install.sh installs these, in order, before it installs
#  opencode itself. The one-line installer already contains this
#  list; this file is the readable reference and is used by the
#  offline (zip) install path.
#
#  One package per line. Text after '#' is a comment.
# ============================================================

curl            # downloads opencode (installer also repairs it if broken)
ca-certificates # HTTPS certificate chain, required for the API and downloads
coreutils       # sha256sum (checksum verification) and standard text tools
tar             # extracts archives; also used for config/project backups
unzip           # extracts zip archives
git             # REQUIRED for /undo and /redo — they are git-backed
ripgrep         # opencode's grep tool will not work without it
nodejs-lts      # node + npm — enables the npm fallback source; not needed by opencode itself
openssh         # OPTIONAL: git over ssh, ssh into the phone. Nothing creates keys automatically.
